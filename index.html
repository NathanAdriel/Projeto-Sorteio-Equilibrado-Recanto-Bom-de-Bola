function exportarPDF() { if (jogadores.length === 0) { alert('Não há jogadores para exportar.'); return; } // 1. Criar uma cópia e ordenar por nota decrescente const ranking = [...jogadores].sort((a, b) => b.nota - a.nota); // 2. Atribuir posições com base na nota (mesma nota = mesma posição, posição sequencial) let posicao = 1; let ultimaNota = null; let posicaoAnterior = 1; const dadosRanking = ranking.map((j, i) => { // Se for o primeiro ou a nota mudou, atualiza a posição if (i === 0 || j.nota < ultimaNota) { posicao = i + 1; ultimaNota = j.nota; posicaoAnterior = posicao; } else { // Mesma nota, mantém a posição anterior posicao = posicaoAnterior; } return { posicao: posicao, nome: j.nome, nota: j.nota, selecionado: j.selecionado ? 'Sim' : 'Não' }; }); // 3. Gerar o PDF const { jsPDF } = window.jspdf; const doc = new jsPDF('p', 'pt', 'a4'); // Título doc.setFontSize(18); doc.text('📊 Ranking de Jogadores - Pelada da Paz', 40, 50); doc.setFontSize(10); doc.text(`Gerado em: ${new Date().toLocaleString()}`, 40, 70); doc.text(`Total de jogadores: ${jogadores.length}`, 40, 85); // Tabela const rows = dadosRanking.map(j => [j.posicao, j.nome, j.nota, j.selecionado]); doc.autoTable({ startY: 100, head: [ ['Posição', 'Nome', 'Nota', 'Selecionado'] ], body: rows, theme: 'striped', headStyles: { fillColor: '#8e44ad', textColor: '#fff', fontSize: 12 }, bodyStyles: { fontSize: 10 }, columnStyles: { 0: { cellWidth: 60, halign: 'center' }, 1: { cellWidth: 200 }, 2: { cellWidth: 60, halign: 'center' }, 3: { cellWidth: 80, halign: 'center' } }, margin: { left: 40, right: 40 } }); // Rodapé const pageCount = doc.internal.getNumberOfPages(); for (let i = 1; i <= pageCount; i++) { doc.setPage(i); doc.setFontSize(8); doc.text(`🕊️ Pelada da Paz - Página ${i} de ${pageCount}`, 40, doc.internal.pageSize.height - 20); } doc.save('ranking_pelada_da_paz.pdf'); }
